4.9. 异步屏障#

异步屏障，引入于 高级同步原语，将 CUDA 同步扩展到 __syncthreads() 或 __syncwarp()，实现细粒度、非阻塞协调以及更好的通信和计算重叠。

本节主要通过以下方式详细介绍如何使用异步屏障 cuda::barrier API（带有指向 cuda::ptx 和原语（如果适用）。

4.9.1. 初始化#

初始化必须在任何线程开始参与屏障之前进行。

#include <cuda/barrier>
#include <cooperative_groups.h>

__global__ void init_barrier()
{
  __shared__ cuda::barrier<cuda::thread_scope_block> bar;
  auto block = cooperative_groups::this_thread_block();

  if (block.thread_rank() == 0)
  {
    // 单个线程初始化总的预期到达计数。
    init(&bar, block.size());
  }
  block.sync();
}

#include <cuda/ptx>
#include <cooperative_groups.h>

__global__ void init_barrier()
{
  __shared__ uint64_t bar;
  auto block = cooperative_groups::this_thread_block();

  if (block.thread_rank() == 0)
  {
    // 单个线程初始化总的预期到达计数。
    cuda::ptx::mbarrier_init(&bar, block.size());
  }
  block.sync();
}

#include <cuda_awbarrier_primitives.h>
#include <cooperative_groups.h>

__global__ void init_barrier()
{
  __shared__ uint64_t bar;
  auto block = cooperative_groups::this_thread_block();

  if (block.thread_rank() == 0)
  {
    // 单个线程初始化总的预期到达计数。
    __mbarrier_init(&bar, block.size());
  }
  block.sync();
}

在任何线程可以参与屏障之前，必须使用以下方法初始化屏障： cuda::barrier::init() 好友功能。这必须在任何线程到达屏障之前发生。这带来了引导挑战，因为线程必须在参与屏障之前同步，但线程正在创建屏障以便同步。在此示例中，将参与的线程是协作组的一部分并使用 block.sync() 引导初始化。由于整个线程块都参与屏障， __syncthreads() 也可以使用。

第二个参数为 init() 是 预计到达人数，即次数 bar.arrive() 在参与线程从其调用中解除阻塞之前，将由参与线程调用 bar.wait(std::move(token))。在此示例和前面的示例中，屏障是使用线程块中的线程数进行初始化的，即 cooperative_groups::this_thread_block().size()，这样线程块内的所有线程都可以参与屏障。

异步屏障可以灵活指定 how 线程参与（分割到达/等待）和 which 线程参与。相比之下， this_thread_block.sync() 或 __syncthreads() 适用于整个线程块并且 __syncwarp(mask) 到warp 的指定子集。尽管如此，如果用户的意图是同步完整的线程块或完整的warp，我们建议使用 __syncthreads() 或 __syncwarp() 分别以获得更好的性能。

4.9.2. 障碍阶段：到达、倒计时、完成和重置#

当参与线程调用时，异步屏障从预期到达计数递减至零 bar.arrive()。当倒计时达到零时，当前阶段的障碍就完成了。当最后一次调用 bar.arrive() 导致倒计时达到零，倒计时会自动自动重置。重置将倒计时分配给预期到达计数，并将障碍移至下一阶段。

A token 类的对象 cuda::barrier::arrival_token，从返回 token=bar.arrive()，与障碍的当前阶段相关。致电 bar.wait(std::move(token)) 当屏障处于当前阶段时，即与令牌关联的阶段与屏障的阶段匹配时，阻塞调用线程。如果在调用之前阶段提前（因为倒计时达到零） bar.wait(std::move(token)) 那么线程不会阻塞；如果在线程被阻塞时阶段提前 bar.wait(std::move(token))，线程被解除阻塞。

必须知道何时可以或不可以发生重置，尤其是在重要的到达/等待同步模式中。

  • 线程的调用 token=bar.arrive() 或 bar.wait(std::move(token)) 必须排序使得 token=bar.arrive() 发生在屏障的当前阶段，并且 bar.wait(std::move(token)) 发生在同一阶段或下一阶段。

  • 一个线程的调用 bar.arrive() 当屏障的计数器非零时必须发生。在屏障初始化之后，如果一个线程调用 bar.arrive() 导致倒计时达到零，然后调用 bar.wait(std::move(token)) 必须在屏障可以重新用于后续调用之前发生 bar.arrive().

  • bar.wait() 只能使用 token 当前阶段或前一阶段的对象。对于任何其他值 token 对象，行为未定义。

对于简单的到达/等待同步模式，遵守这些使用规则很简单。

4.9.2.1. warp 纠缠#

warp 发散会影响到达操作更新障碍的次数。如果调用warp完全收敛，则屏障更新一次。如果调用的warp完全发散，则 32 个单独的更新将应用于屏障。

注

建议 arrive-on(bar) 聚合线程使用调用来最小化对屏障对象的更新。当这些操作之前的代码使线程发散时，应该重新聚合warp，通过 __syncwarp 在调用到达操作之前。

4.9.3. 显式相位跟踪#

异步屏障可以有多个阶段，具体取决于它用于同步线程和内存操作的次数。我们可以使用以下方法直接跟踪阶段，而不是使用令牌来跟踪势垒阶段翻转 mbarrier_try_wait_parity() 可通过以下方式获得的函数系列 cuda::ptx 和原语 API。

以最简单的形式， cuda::ptx::mbarrier_try_wait_parity(uint64_t* bar, const uint32_t& phaseParity) 函数等待具有特定奇偶校验的阶段。的 phaseParity 操作数是屏障对象的当前阶段或前一阶段的整数奇偶校验。偶数相位的整数奇偶校验为 0，奇数相位的整数奇偶校验为 1。当我们初始化屏障时，其相位的奇偶校验为 0。因此 phaseParity 是 0 和 1。在跟踪时显式相位跟踪非常有用 异步内存操作，因为它只允许单个线程到达屏障并设置事务计数，而其他线程仅等待基于奇偶校验的相位翻转。这比让所有线程到达屏障并使用令牌更有效。此功能仅适用于线程块和集群范围内的共享内存屏障。

#include <cuda/ptx>
#include <cooperative_groups.h>

__device__ void compute(float *data, int iteration);

__global__ void split_arrive_wait(int iteration_count, float *data)
{
  using barrier_t = cuda::barrier<cuda::thread_scope_block>;
  __shared__ barrier_t bar;
  int parity = 0; // 初始相位奇偶校验为0。
  auto block = cooperative_groups::this_thread_block();

  if (block.thread_rank() == 0)
  {
    // 使用预期到达计数初始化障碍。
    init(&bar, block.size());
  }
  block.sync();

  for (int i = 0; i < iteration_count; ++i)
  {
    /* 到达前的代码 */

    // 该线程到达。到达不会阻塞线程。
    // 获取与 cuda::ptx API 一起使用的本机屏障的句柄。
    (void)cuda::ptx::mbarrier_arrive(cuda::device::barrier_native_handle(bar));

    compute(data, i);

    // 等待参与屏障的所有线程完成 mbarrier_arrive()。
    // 获取与 cuda::ptx API 一起使用的本机屏障的句柄。
    while (!cuda::ptx::mbarrier_try_wait_parity(cuda::device::barrier_native_handle(bar), parity)) {}
    // 翻转平价。
    parity ^= 1;

    /* 等待后的代码 */
  }
}

#include <cuda/ptx>
#include <cooperative_groups.h>

__device__ void compute(float *data, int iteration);

__global__ void split_arrive_wait(int iteration_count, float *data)
{
  __shared__ uint64_t bar;
  int parity = 0; // 初始相位奇偶校验为0。
  auto block = cooperative_groups::this_thread_block();

  if (block.thread_rank() == 0)
  {
    // 使用预期到达计数初始化障碍。
    cuda::ptx::mbarrier_init(&bar, block.size());
  }
  block.sync();

  for (int i = 0; i < iteration_count; ++i)
  {
    /* 到达前的代码 */

    // 该线程到达。到达不会阻塞线程。
    (void)cuda::ptx::mbarrier_arrive(&bar);

    compute(data, i);

    // 等待参与屏障的所有线程完成 mbarrier_arrive()。
    while (!cuda::ptx::mbarrier_try_wait_parity(&bar, parity)) {}
    // 翻转平价。
    parity ^= 1;

    /* 等待后的代码 */
  }
}

#include <cuda_awbarrier_primitives.h>
#include <cooperative_groups.h>

__device__ void compute(float *data, int iteration);

__global__ void split_arrive_wait(int iteration_count, float *data)
{
  __shared__ __mbarrier_t bar;
  bool parity = false; // 初始相位奇偶校验为假。
  auto block = cooperative_groups::this_thread_block();

  if (block.thread_rank() == 0)
  {
    // 使用预期到达计数初始化障碍。
    __mbarrier_init(&bar, block.size());
  }
  block.sync();

  for (int i = 0; i < iteration_count; ++i)
  {
    /* 到达前的代码 */

    (void)__mbarrier_arrive(&bar);

    compute(data, i);

    // 等待参与屏障的所有线程完成 __mbarrier_arrive()。
    while(!__mbarrier_try_wait_parity(&bar, parity, 1000)) {}
    parity ^= 1;

    /* 等待后的代码 */
  }
}

4.9.4. 提前退出#

当参与同步序列的线程必须提前退出该序列时，该线程必须在退出之前显式退出参与。其余参与线程可以正常进行后续的到达和等待操作。

#include <cuda/barrier>
#include <cooperative_groups.h>

__device__ bool condition_check();

__global__ void early_exit_kernel(int N)
{
  __shared__ cuda::barrier<cuda::thread_scope_block> bar;
  auto block = cooperative_groups::this_thread_block();

  if (block.thread_rank() == 0)
  {
    init(&bar, block.size());
  }
  block.sync();

  for (int i = 0; i < N; ++i)
  {
    if (condition_check())
    {
      bar.arrive_and_drop();
      return;
    }
    // 其他线程可以正常进行。
    auto token = bar.arrive();

    /* 到达和等待之间的代码 */

    // 等待所有线程到达。
    bar.wait(std::move(token));

    /* 等待后的代码 */
  }
}

#include <cuda_awbarrier_primitives.h>
#include <cooperative_groups.h>

__device__ bool condition_check();

__global__ void early_exit_kernel(int N)
{
  __shared__ __mbarrier_t bar;
  auto block = cooperative_groups::this_thread_block();

  if (block.thread_rank() == 0)
  {
    __mbarrier_init(&bar, block.size());
  }
  block.sync();

  for (int i = 0; i < N; ++i)
  {
    if (condition_check())
    {
      __mbarrier_token_t token = __mbarrier_arrive_and_drop(&bar);
      return;
    }
    // 其他线程可以正常进行。
    __mbarrier_token_t token = __mbarrier_arrive(&bar);

    /* 到达和等待之间的代码 */

    // 等待所有线程到达。
    while (!__mbarrier_try_wait(&bar, token, 1000)) {}

    /* 等待后的代码 */
  }
}

bar.arrive_and_drop() 操作到达屏障以履行参与线程到达屏障的义务 current 阶段，然后减少预期到达计数 next 阶段，以便该线程不再预期到达屏障。

4.9.5. 完成功能#

cuda::barrier API 支持可选的完成功能。一个 CompletionFunction 或 cuda::barrier<Scope, CompletionFunction> 每个阶段在最后一个线程之后执行一次 arrives 并且在任何线程被解除阻塞之前 wait。到达的线程执行的内存操作 barrier 在此阶段期间对执行该阶段的线程可见 CompletionFunction，以及在 CompletionFunction 对所有等待的线程可见 barrier 一旦他们被解除封锁 wait.

#include <cuda/barrier>
#include <cooperative_groups.h>
#include <functional>
namespace cg = cooperative_groups;

__device__ int divergent_compute(int *, int);
__device__ int independent_computation(int *, int);

__global__ void psum(int *data, int n, int *acc)
{
  auto block = cg::this_thread_block();

  constexpr int BlockSize = 128;
  __shared__ int smem[BlockSize];
  assert(BlockSize == block.size());
  assert(n % BlockSize == 0);

  auto completion_fn = [&]
  {
    int sum = 0;
    for (int i = 0; i < BlockSize; ++i)
    {
      sum += smem[i];
    }
    *acc += sum;
  };

  /* 屏障存储。
     注意：屏障不可默认构造，因为
           由于completion_fn不可默认构造
           到捕获。 */
  using completion_fn_t = decltype(completion_fn);
  using barrier_t = cuda::barrier<cuda::thread_scope_block,
                                  completion_fn_t>;
  __shared__ std::aligned_storage<sizeof(barrier_t),
                                  alignof(barrier_t)>
      bar_storage;

  // 初始化屏障。
  barrier_t *bar = (barrier_t *)&bar_storage;
  if (block.thread_rank() == 0)
  {
    assert(*acc == 0);
    assert(blockDim.x == blockDim.y == blockDim.y == 1);
    new (bar) barrier_t{block.size(), completion_fn};
    /* 相当于： init(bar, block.size(),completion_fn); */
  }
  block.sync();

  // 主循环。
  for (int i = 0; i < n; i += block.size())
  {
    smem[block.thread_rank()] = data[i] + *acc;
    auto token = bar->arrive();
    // 我们可以在这里进行独立计算。
    bar->wait(std::move(token));
    // 共享内存可以在下一次迭代中安全地重用
    // 因为所有线程都用它完成，包括一个
    // 这确实减少了。
  }
}

4.9.6. 跟踪异步内存操作#

异步屏障可用于跟踪 异步内存复制。当异步复制操作绑定到屏障时，复制操作会在启动时自动增加当前屏障阶段的预期计数，并在完成时减少它。这种机制确保了屏障的 wait() 操作将阻塞，直到所有关联的异步内存复制完成为止，从而提供了一种同步多个并发内存操作的便捷方法。

从计算能力 9.0 开始，具有线程块或集群范围的共享内存中的异步屏障可以 显式地 跟踪异步内存操作。我们将这些障碍称为 异步事务障碍。除了预期到达计数之外，屏障对象还可以接受 交易数量，可用于跟踪异步事务的完成情况。事务计数跟踪未完成且尚未完成的异步事务的数量，以异步内存操作指定的单位（通常为字节）。当前阶段要跟踪的事务计数可以在到达时设置 cuda::device::barrier_arrive_tx() 或直接与 cuda::device::barrier_expect_tx()。当屏障使用事务计数时，它会在等待操作时阻塞线程，直到所有生产者线程都执行了到达操作 — 所有交易计数的总和达到预期值。

#include <cuda/barrier>
#include <cooperative_groups.h>

__global__ void track_kernel()
{
  __shared__ cuda::barrier<cuda::thread_scope_block> bar;
  auto block = cooperative_groups::this_thread_block();

  if (block.thread_rank() == 0)
  {
    init(&bar, block.size());
  }
  block.sync();

  auto token = cuda::device::barrier_arrive_tx(bar, 1, 0);

  bar.wait(cuda::std::move(token));
}

#include <cuda/ptx>
#include <cooperative_groups.h>

__global__ void track_kernel()
{
  __shared__ uint64_t bar;
  auto block = cooperative_groups::this_thread_block();

  if (block.thread_rank() == 0)
  {
    cuda::ptx::mbarrier_init(&bar, block.size());
  }
  block.sync();

  uint64_t token = cuda::ptx::mbarrier_arrive_expect_tx(cuda::ptx::sem_release, cuda::ptx::scope_cluster, cuda::ptx::space_shared, &bar, 1, 0);

  while (!cuda::ptx::mbarrier_try_wait(&bar, token)) {}
}

在此示例中， cuda::device::barrier_arrive_tx() 操作构造与当前阶段的阶段同步点关联的到达令牌对象。然后，将到达计数递减 1，并将预期事务计数递增 0。由于事务计数更新为 0，因此屏障不跟踪任何事务。后续部分将介绍 使用张量内存加速器（TMA） 包括跟踪异步内存操作的示例。

4.9.7. 使用壁垒的生产者-消费者模式#

线程块可以进行空间分区，以允许不同的线程执行独立的操作。这通常是通过将线程块内不同线程束的线程分配给特定任务来完成的。该技术被称为 warp 专门化.

本节展示了生产者-消费者模式中的空间分区示例，其中一个线程子集生成由另一（不相交）线程子集同时使用的数据。生产者-消费者空间分区模式需要两个单侧同步来管理生产者和消费者之间的数据缓冲区。

#include <cuda/barrier>

using barrier_t = cuda::barrier<cuda::thread_scope_block>;

__device__ void produce(barrier_t ready[], barrier_t filled[], float *buffer, int buffer_len, float *in, int N)
{
  for (int i = 0; i < N / buffer_len; ++i)
  {
    ready[i % 2].arrive_and_wait(); /* 等待 buffer_(i%2) 准备好被填充 */
    /* 产生，即填充，buffer_(i%2) */
    barrier_t::arrival_token token = filled[i % 2].arrive(); /* buffer_(i%2) 已满 */
  }
}

__device__ void consume(barrier_t ready[], barrier_t filled[], float *buffer, int buffer_len, float *out, int N)
{
  barrier_t::arrival_token token1 = ready[0].arrive(); /* buffer_0 已准备好进行初始填充 */
  barrier_t::arrival_token token2 = ready[1].arrive(); /* buffer_1 已准备好进行初始填充 */
  for (int i = 0; i < N / buffer_len; ++i)
  {
    filled[i % 2].arrive_and_wait(); /* 等待 buffer_(i%2) 被填满 */
    /* 消耗缓冲区_(i%2) */
    barrier_t::arrival_token token3 = ready[i % 2].arrive(); /* buffer_(i%2) 已准备好重新填充 */
  }
}

__global__ void producer_consumer_pattern(int N, float *in, float *out, int buffer_len)
{
  constexpr int warpSize = 32;

  /* 下面声明的共享内存缓冲区的大小为 2 * buffer_len
     这样我们就可以在两个缓冲区之间交替工作。
     buffer_0 = 缓冲区且 buffer_1 = 缓冲区 + buffer_len */
  __shared__ extern float buffer[];

  /* bar[0] 和 bar[1] 跟踪缓冲区 buffer_0 和 buffer_1 是否已准备好填充，
     而 bar[2] 和 bar[3] 分别跟踪缓冲区 buffer_0 和 buffer_1 是否已填充 */
  #pragma nv_diag_suppress static_var_with_dynamic_init
  __shared__ barrier_t bar[4];

  if (threadIdx.x < 4)
  {
    init(bar + threadIdx.x, blockDim.x);
  }
  __syncthreads();

  if (threadIdx.x < warpSize)
  { produce(bar, bar + 2, buffer, buffer_len, in, N); }
  else
  { consume(bar, bar + 2, buffer, buffer_len, out, N); }
}

在此示例中，第一个 warp 专门为生产者，其余 warp 专门为消费者。所有生产者和消费者线程都参与（调用 bar.arrive() 或 bar.arrive_and_wait()）在四个障碍中的每一个中，因此预期到达计数等于 block.size().

生产者线程等待消费者线程发出共享内存缓冲区可以被填充的信号。为了等待屏障，生产者线程必须首先到达该屏障 ready[i%2].arrive() 获得一个令牌，然后 ready[i%2].wait(token) 用那个令牌。为了简单起见， ready[i%2].arrive_and_wait() 结合了这些操作。

bar.arrive_and_wait();
/* 相当于 */
bar.wait(bar.arrive());

生产者线程计算并填充就绪缓冲区，然后它们通过到达已填充的屏障来发出缓冲区已填充的信号， filled[i%2].arrive()。生产者线程此时不会等待，而是等待直到下一次迭代的缓冲区（双缓冲）准备好被填充。

消费者线程首先发出两个缓冲区已准备好填充的信号。消费者线程此时不会等待，而是等待本次迭代的缓冲区被填充， filled[i%2].arrive_and_wait()。消费者线程消耗完缓冲区后，它们发出信号表明缓冲区已准备好再次填充， ready[i%2].arrive()，然后等待下一次迭代的缓冲区被填满。
